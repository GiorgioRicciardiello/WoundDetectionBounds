\begin{table}[htbp]
	\centering
	\caption{Per-frame processing sequence of the temporal integrator ($t > 0$).}
	\label{tab:temporal-sequence}
	
	\begin{tabular}{
			>{\centering\arraybackslash}m{0.08\linewidth}
			>{\arraybackslash}m{0.52\linewidth}
			>{\arraybackslash}m{0.30\linewidth}
		}
		\toprule
		Step & Operation & Implemented in \\
		\midrule
		
		1  & Single-frame detector $\rightarrow$ raw edges $u_\text{det}(x,t)$, $\ell_\text{det}(x,t)$ 
		& Section~2.3.2 \\
		
		2  & NaN guard: replace non-finite detector outputs with previous state 
		& \texttt{KalmanEdgeFilter.update} \\
		
		3  & Spatial Gaussian smoothing of detected edges ($\sigma_\text{spatial} = 5$ columns) 
		& \texttt{gaussian\_filter1d} \\
		
		4  & Per-column observation noise $R(x,t)$ estimation 
		& \texttt{estimate\_R\_per\_column} \\
		
		5  & Kalman predict step: $P^-(x,t) = P(x,t-1) + Q$ 
		& \texttt{\_kalman\_step} \\
		
		6  & Kalman update: gain $K$, blended edge $u_\text{filt}$, contracted $P_\text{upd}$ 
		& \texttt{\_kalman\_step} \\
		
		7  & NaN guard on Kalman output: reset non-finite $u_\text{filt}$ to $u(x,t-1)$ 
		& \texttt{KalmanEdgeFilter.update} \\
		
		8  & Hard safety net: $u(x,t) = \max(u_\text{filt}, u(x,t-1))$ 
		& \texttt{KalmanEdgeFilter.update} \\
		
		9  & P-state correction: reset $P$ to $P^-$ for columns where safety net fired 
		& \texttt{KalmanEdgeFilter.update} \\
		
		10 & Mask reconstruction from constrained edges 
		& \texttt{\_edges\_to\_mask} \\
		
		11 & Last-resort area backstop: emit warning and revert if $A(t) > A(t-1)$ 
		& \texttt{QuantificationSegmenter.segment} \\
		
		\bottomrule
	\end{tabular}
\end{table}
